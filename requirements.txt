# Hybrid IDS — runtime dependencies (Windows, macOS, Linux).
#
#   python -m pip install -r requirements.txt
#   python run_hybrid_ids.py            (as Administrator / sudo, for capture)
#
# These are the exact versions the project is built and verified against
# (Python 3.14.5). They are pinned rather than left floating so a fresh install
# reproduces a known-good environment instead of pulling whatever is latest on
# the day.
#
# scikit-learn is the load-bearing pin: the shipped models (rf_model.pkl,
# rf_scaler.pkl, ids_model_live.pkl) are pickled with _sklearn_version 1.8.0.
# Loading them under a different scikit-learn raises InconsistentVersionWarning
# and can fail to unpickle — which would silently drop the Random Forest and
# fall back to the K-Means model. Retrain the models if you change this pin.
#
# streamlit is pinned because the dashboard is verified on 1.57.0. Streamlit
# removes deprecated APIs on a schedule, and the dashboard still calls
# st.components.v1.html in ~8 places (the FX layer, the Educational Simulator
# and the Leaflet threat map are all self-contained iframes). That API is past
# its announced removal date and warns on 1.57 — it will stop working on some
# future release, so an unpinned upgrade is a live breakage risk. Migrating
# those call sites to st.iframe is the open follow-up.
#
# To upgrade deliberately: bump one pin, then launch the dashboard and confirm
# every tab still renders.
#
# On an older Python (3.11/3.12) some of these wheels do not exist. Drop the
# pins to floors instead — streamlit>=1.40, pandas>=2.0, numpy>=1.24,
# scikit-learn>=1.3, joblib>=1.3, plotly>=5.18, scapy>=2.5, requests>=2.31,
# python-docx>=1.1, matplotlib>=3.7 — and retrain the models if scikit-learn
# resolves to anything other than 1.8.0.

# ── Runtime (dashboard + detection engine) ────────────────────────────────────
streamlit==1.57.0
pandas==3.0.2
numpy==2.4.4
scikit-learn==1.8.0
joblib==1.5.3
plotly==6.8.0          # PCAP Analysis + Threat Map charts
scapy==2.7.0           # Rui Yang PCAP engine
requests==2.33.1       # Rui Yang geoip lookups (PCAP Analysis + Threat Map)
psutil>=5.9            # live_backend network-interface auto-detect
python-docx==1.2.0     # Word export for the Threat/Management reports
matplotlib==3.10.9     # SHAP plots + report charts

# ── Model Intelligence (Megan): LSTM sequence layer + SHAP explainability ─────
# These are runtime dependencies, not extras. torch is what
# live_backend._load_lstm_safe() loads: without it the detection engine quietly
# runs one layer short (sig_lstm never populates) and the fused threat level is
# missing the sequence model's vote. shap drives both explainability panels,
# which otherwise show an "install shap" message instead of a chart. Leaving
# them commented out is what sent teammates hunting for them mid-demo.
torch==2.12.0          # LSTM sequence model (CPU build is enough)
shap==0.52.0           # SHAP explainability (RF TreeExplainer + LSTM Gradient)
